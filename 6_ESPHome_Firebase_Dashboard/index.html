📚 ข้อกำหนดเซนเซอร์ นศ. (ขั้นต่ำ 6 คอลัมน์)

นักศึกษากำหนดชื่อคอลัมน์เซนเซอร์ตามโจทย์ของตนเองได้อิสระ (อย่างน้อย 6 คอลัมน์) ตัวอย่าง JSON ที่ส่งไปยัง /latest.json:

{
  "temp": 28.5,       // อุณหภูมิอากาศ (°C)
  "humi": 65.0,       // ความชื้นสัมพัทธ์ (%)
  "light": 520,       // ความเข้มแสง (lx)
  "soil": 60.0,       // ความชื้นในดิน (%)
  "pm25": 14.5,       // ฝุ่น PM2.5 (µg/m³)
  "pressure": 1012.3, // ความกดอากาศ (hPa)
  "timestamp": 1740000000
}
